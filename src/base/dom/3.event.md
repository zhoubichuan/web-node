# 文件

## 1.分割文件

::: demo

```vue
<template>
  <div class="fs-box">
      根目录：<div>{{path}}</div>
      文件夹：<div>{{foldName}}</div>
     <input type="text" v-model="count" style="width: calc(100% - 100px);"  />
    <button @click="handleList">获取根目录</button>
    根目录：{{fold}}
  </div>
</template>
<script>
export default {
  data() {
    return {
      fold: '',
      path: '',
      foldName:[],
      fileType:[],
      count:6
    }
  },
  mounted() {
    // this.handleList()
   this.path = localStorage.getItem('path')
    localStorage.getItem('foldName')&& this.foldName.splice(0,this.foldName.length,...localStorage.getItem('foldName').split(','))
  },
  methods: {
    async handleList(){
      const res = await this.$api.file({ type: 'move', path: this.path ,foldName:[this.foldName],count:this.count})
      if (!res) return
      this.fold = res
    }
  }
}
</script>
```

## 2.文件类型设置

::: demo

```vue
<template>
  <div class="fs-box">
    <button @click=handleClear()>清空</button>
     <button v-for="(type,index) in typeList" :key="index" @click=handleClick(type)>{{type}} </button>
     <input type="text" v-model="value"  style="width: calc(100% - 100px);" /><button @click=handleClick(value)>确定</button>
      根目录：<div>{{path}}</div>
      文件夹：<div>{{foldName}}</div>
      文件类型：<div>{{fileType}}</div>
  </div>
</template>
<script>
export default {
  data() {
    return {
      typeList:[
        'png',
        'jpg',
        'jpeg',
        'gif',
        'webp',
        'svg',
      ],
      value:'',
      path:'',
      foldName:[],
      fileType:[],
    }
  },
  mounted() {
  localStorage.getItem('fileType')&& this.fileType.splice(0,this.fileType.length,...localStorage.getItem('fileType').split(','))
  },
  methods: {
    handleClear(){
      this.fileType=[]
      this.value = ''
      localStorage.setItem('fileType','')
    },
    handleClick(item){
      this.fileType.push(item)
      localStorage.setItem('fileType',this.fileType?.join(','))
    }
  }
}
</script>
```
