# 文件夹

## 1.文件夹下文件列表顺序修改

::: demo

```vue
<template>
  <div class="wrap">
    <button @click="handleList">获取列表</button>
      根目录：<div>{{items.path}}</div>
      文件夹：<div>{{items.foldName}}</div>
      文件类型：<div>{{items.fileType}}</div>
    <div class="box-images">
      <div class="item" v-for="item in items.fileName" :draggable="true" @dragstart="dragstart(item)" @drop="drop(item)"
        @dragover="dragover" :key="item.name">
        {{ item.name }}
        <img :src="link + items.foldName[0]+'/' + item.name + '.' + items.fileType[0]">
      </div>
    </div>
  </div>
</template>
<script>
export default {
  data() {
    return {
      link:'',
      items:{
        path: "",
        foldName: [],
        fileType: [],
        type: 'edit',
        fileName: []
      },
      preIndex:0
    }
  },
  methods: {
   async handleList() {
      const files = await this.$api.file({...this.items, type: 'read',fileName:undefined})
      this.items.fileName = files.map((item, index) => {
        return {
          name: item.split('.')[0],
          rename: index + 1,
          index: index + 1,
        }
      })
    },
    dragstart(item) {
      this.preIndex = item;
    },
    async drop (item){
      this.items.fileName.splice(this.preIndex.index - 1, 1, {
        ...item,
        index: this.preIndex.index,
        rename: this.preIndex.index,
      });
      this.items.fileName.splice(item.index - 1, 1, {
        ...this.preIndex,
        index: item.index,
        rename: item.index,
      });
      this.preIndex.index = 0 
      await this.$api.file(JSON.parse(JSON.stringify(this.items)))
      await this.handleList()
    },
    dragover (e) {
      e.preventDefault();
    },
  },
  mounted() {
    // this.handleList()
      this.link = localStorage.getItem('link')
       this.items.path = localStorage.getItem('path')
       localStorage.getItem('foldName')&& this.items.foldName.splice(0,this.items.foldName.length,...localStorage.getItem('foldName').split(','))
      localStorage.getItem('fileType')&& this.items.fileType.splice(0,this.items.fileType.length,...localStorage.getItem('fileType').split(','))
  },
}
</script>

  background-color: lightcyan;
  float: left;
}

  max-height: 100%;
  max-width: 100%;
}
</style>
```
